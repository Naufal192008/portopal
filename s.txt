<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, ...$roles)
    {
        if (!auth()->check()) {
            return redirect()->route('login');
        }

        if (!in_array(auth()->user()->role, $roles)) {
            abort(403, 'Akses ditolak. Role Anda tidak diizinkan.');
        }

        return $next($request);
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;

class AuthController extends Controller
{
    // ============ LOGIN ============
    public function showLogin()
    {
        if (Auth::check()) {
            return redirect()->route('dashboard');
        }
        return view('auth.login');
    }

    public function login(Request $request)
    {
        $request->validate([
            'username' => 'required',
            'password' => 'required',
        ]);

        $user = User::where('username', $request->username)->first();

        if (!$user || !Hash::check($request->password, $user->password)) {
            return back()->withErrors(['username' => 'Username atau password salah!'])->withInput();
        }

        if (!$user->is_active) {
            return back()->withErrors(['username' => 'Akun Anda dinonaktifkan!'])->withInput();
        }

        Auth::login($user);
        $request->session()->regenerate();

        return redirect()->route('dashboard')->with('success', 'Selamat datang, ' . $user->name . '!');
    }

    // ============ REGISTER ============
    public function showRegister()
    {
        if (Auth::check()) {
            return redirect()->route('dashboard');
        }
        return view('auth.register');
    }

    public function register(Request $request)
    {
        $request->validate([
            'name' => 'required|min:3|max:100',
            'username' => 'required|min:3|max:50',
            'password' => 'required|min:6|confirmed',
            'role' => 'required|in:owner,administrator,waiter,kasir',
        ]);

        try {
            $hash = Hash::make($request->password);

            DB::statement("CALL sp_register_user(?, ?, ?, ?)", [
                $request->name,
                $request->username,
                $hash,
                $request->role,
            ]);

            return redirect()->route('login')->with('success', 'Registrasi berhasil! Silakan login.');
        } catch (\Exception $e) {
            return back()->withErrors(['username' => $e->getMessage()])->withInput();
        }
    }

    // ============ LOGOUT ============
    public function logout(Request $request)
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();
        return redirect()->route('login');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\MenuItem;
use Illuminate\Http\Request;

class MenuController extends Controller
{
    public function index()
    {
        $items = MenuItem::orderBy('category')->orderBy('item_name')->paginate(10);
        return view('menu.index', compact('items'));
    }

    public function create()
    {
        return view('menu.form');
    }

    public function store(Request $request)
    {
        $request->validate([
            'item_code' => 'required|unique:menu_items',
            'item_name' => 'required|min:3',
            'price' => 'required|numeric|min:0',
        ]);

        MenuItem::create([
            'item_code' => $request->item_code,
            'item_name' => $request->item_name,
            'category' => $request->category,
            'price' => $request->price,
            'is_available' => $request->has('is_available') ? 1 : 0,
        ]);

        return redirect()->route('menu.index')->with('success', 'Menu berhasil ditambahkan!');
    }

    public function show(MenuItem $menu)
    {
        return view('menu.show', compact('menu'));
    }

    public function edit(MenuItem $menu)
    {
        return view('menu.form', compact('menu'));
    }

    public function update(Request $request, MenuItem $menu)
    {
        $request->validate([
            'item_code' => 'required|unique:menu_items,item_code,' . $menu->id,
            'item_name' => 'required|min:3',
            'price' => 'required|numeric|min:0',
        ]);

        $menu->update([
            'item_code' => $request->item_code,
            'item_name' => $request->item_name,
            'category' => $request->category,
            'price' => $request->price,

<?php

namespace App\Http\Controllers;

use App\Models\MenuItem;
use App\Models\Order;
use App\Models\RestoTable;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class OrderController extends Controller
{
    public function index()
    {
        $tables = RestoTable::where('status', 'available')->get();
        $menuItems = MenuItem::where('is_available', true)->orderBy('category')->get();
        $activeOrders = Order::with(['table', 'user'])->where('status', '!=', 'paid')->latest()->get();

        return view('orders.index', compact('tables', 'menuItems', 'activeOrders'));
    }

    public function store(Request $request)
    {
        $request->validate(['table_id' => 'required|exists:resto_tables,id']);

        DB::beginTransaction();
        try {
            $order = Order::create([
                'table_id' => $request->table_id,
                'user_id' => auth()->id(),
                'status' => 'pending',
            ]);

            RestoTable::where('id', $request->table_id)->update(['status' => 'occupied']);

            DB::commit();
            return redirect()->route('orders.show', $order)->with('success', 'Order berhasil dibuat! Silakan tambah item.');
        } catch (\Exception $e) {
            DB::rollBack();
            return back()->with('error', 'Gagal membuat order: ' . $e->getMessage());
        }
    }

    public function show(Order $order)
    {
        $order->load(['table', 'user', 'items.menuItem']);
        $menuItems = MenuItem::where('is_available', true)->orderBy('category')->get();
        return view('orders.show', compact('order', 'menuItems'));
    }

    public function addItem(Request $request, Order $order)
    {
        $request->validate([
            'menu_item_id' => 'required|exists:menu_items,id',
            'quantity' => 'required|integer|min:1',
        ]);

        try {
            DB::statement("CALL sp_add_order_item(?, ?, ?, ?)", [
                $order->id,
                $request->menu_item_id,
                $request->quantity,
                $request->note ?? '',
            ]);

            return back()->with('success', 'Item berhasil ditambahkan!');
        } catch (\Exception $e) {
            return back()->with('error', 'Gagal: ' . $e->getMessage());
        }
    }

    public function destroy(Order $order)
    {
        RestoTable::where('id', $order->table_id)->update(['status' => 'available']);
        $order->delete();
        return redirect()->route('orders.index')->with('success', 'Order dibatalkan!');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Order;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class TransaksiController extends Controller
{
    public function index()
    {
        $activeOrders = Order::with(['table', 'user'])->where('status', '!=', 'paid')->latest()->get();
        return view('transaksi.index', compact('activeOrders'));
    }

    public function proses(Request $request)
    {
        $request->validate([
            'order_id' => 'required|exists:orders,id',
            'payment_method' => 'required|in:cash,credit_card,debit_card,qr_code',
            'amount_paid' => 'required|numeric|min:0',
        ]);

        $order = Order::findOrFail($request->order_id);

        if ($request->amount_paid < $order->total) {
            return back()->with('error', 'Jumlah bayar kurang dari total!');
        }

        try {
            DB::statement("CALL sp_complete_transaction(?, ?, ?, ?)", [
                $order->id,
                auth()->id(),
                $request->payment_method,
                $request->amount_paid,
            ]);

            $kembalian = $request->amount_paid - $order->total;
            return redirect()->route('transaksi.index')->with('success', 'Pembayaran berhasil! Kembalian: Rp ' . number_format($kembalian, 0, ',', '.'));
        } catch (\Exception $e) {
            return back()->with('error', 'Gagal: ' . $e->getMessage());
        }
    }
}

        } catch (\Exception $e) {

        $totalTransactions = count($transactions);
