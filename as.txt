**Bukti — Test manual:**
- Login owner → buka `/barang` → **BISA AKSES** ❌

✅ **Analisis BENAR** — tidak ada middleware role.

---

#### 5️⃣ PERBAIKAN

**1. Buat middleware:**
```cmd
php artisan make:middleware RoleMiddleware
```

**2. Isi `RoleMiddleware.php`:**
```php
<?php
namespace App\Http\Middleware;
use Closure;
use Illuminate\Http\Request;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, ...$roles)
    {
        if (!auth()->check()) {
            return redirect()->route('login');
        }
        if (!in_array(auth()->user()->role, $roles)) {
            abort(403, 'Akses ditolak.');
        }
        return $next($request);
    }
}
```

**3. Daftarkan di `bootstrap/app.php`:**
```php
->withMiddleware(function (Middleware $middleware) {
    $middleware->alias([
        'role' => \App\Http\Middleware\RoleMiddleware::class,
    ]);
})
```

**4. Update `routes/web.php`:**
```php
Route::middleware('role:owner,administrator,petugas')->group(function () {
    Route::get('/', [DashboardController::class, 'index'])->name('dashboard');
});
```

```sql
CREATE PROCEDURE sp_tambah_penawaran(
    IN p_id_lelang INT,
    IN p_id_barang INT,
    IN p_id_user INT,
    IN p_penawaran INT
)
BEGIN
    DECLARE v_harga_tertinggi INT;
    DECLARE v_status VARCHAR(10);

    SELECT status INTO v_status FROM tb_lelang WHERE id_lelang = p_id_lelang;
    IF v_status = 'ditutup' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Lelang sudah ditutup!';
    END IF;

    SELECT MAX(penawaran_harga) INTO v_harga_tertinggi
    FROM tb_history_lelang WHERE id_lelang = p_id_lelang;

    IF v_harga_tertinggi IS NOT NULL AND p_penawaran <= v_harga_tertinggi THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Penawaran harus lebih tinggi!';
    END IF;

    INSERT INTO tb_history_lelang (id_lelang, id_barang, id_user, penawaran_harga)
    VALUES (p_id_lelang, p_id_barang, p_id_user, p_penawaran);
END
```

```sql
CREATE PROCEDURE sp_tutup_lelang(IN p_id_lelang INT)
BEGIN
    DECLARE v_harga_akhir INT;
    DECLARE v_id_user INT;

    SELECT id_user, penawaran_harga INTO v_id_user, v_harga_akhir
    FROM tb_history_lelang
    WHERE id_lelang = p_id_lelang
    ORDER BY penawaran_harga DESC LIMIT 1;

    UPDATE tb_lelang
    SET status = 'ditutup', harga_akhir = v_harga_akhir, id_user = v_id_user
    WHERE id_lelang = p_id_lelang;
END
```
