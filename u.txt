<?php
// ============================================
// DATABASE.PHP - KONEKSI DAN HELPER FUNCTIONS
// ============================================

if (session_status() == PHP_SESSION_NONE) {
    session_start();
}

class Database {
    private static $instance = null;
    private $pdo;
    private $host = 'localhost';
    private $dbname = 'restaurant_pos';
    private $username = 'root';
    private $password = '';
    
    private function __construct() {
        try {
            $this->pdo = new PDO(
                "mysql:host={$this->host};dbname={$this->dbname};charset=utf8mb4",
                $this->username,
                $this->password
            );
            $this->pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
            $this->pdo->setAttribute(PDO::ATTR_DEFAULT_FETCH_MODE, PDO::FETCH_ASSOC);
        } catch (PDOException $e) {
            die("❌ Koneksi database gagal: " . $e->getMessage());
        }
    }
    
    public static function getInstance() {
        if (self::$instance == null) {
            self::$instance = new Database();
        }
        return self::$instance;
    }
    
    public function getConnection() {
        return $this->pdo;
    }
    
    public function getOne($sql, $params = []) {
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt->fetch();
    }
    
    public function getAll($sql, $params = []) {
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt->fetchAll();
    }
    
    public function execute($sql, $params = []) {
        $stmt = $this->pdo->prepare($sql);
        return $stmt->execute($params);
    }
    
    public function insert($sql, $params = []) {
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $this->pdo->lastInsertId();
    }
    
    // ============================================
    // STORED PROCEDURE
    // ============================================
    public function callProcedure($name, $params = []) {
        $placeholders = implode(',', array_fill(0, count($params), '?'));
        $sql = "CALL $name($placeholders)";
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        
        $results = [];
        do {
            $results[] = $stmt->fetchAll();
        } while ($stmt->nextRowset());
        
        return $results;
    }
}

// ============================================
// HELPER FUNCTIONS
// ============================================

function isLoggedIn() {
    return isset($_SESSION['user_id']) && $_SESSION['user_id'] > 0;
}

function getCurrentUser() {
    return $_SESSION['full_name'] ?? null;
}

function getCurrentRole() {
    return $_SESSION['role'] ?? null;
}

function getCurrentUserId() {
    return $_SESSION['user_id'] ?? 0;
}

function requireLogin() {
    if (!isLoggedIn()) {
        header('Location: index.php?page=login');
        exit;
    }
}

function requireRole($role) {
    requireLogin();
    $currentRole = getCurrentRole();
    if ($currentRole != $role && $currentRole != 'administrator') {
        header('Location: index.php?page=dashboard');
        exit;
    }
}

function hashPassword($password) {
    return md5($password);
}

function formatRupiah($amount) {
    return 'Rp ' . number_format($amount, 0, ',', '.');
}

<?php
// ============================================
// AUTHCONTROLLER.PHP - LOGIN/LOGOUT
// ============================================

require_once __DIR__ . '/../models/User.php';

class AuthController {
    private $user;
    
    public function __construct() {
        $this->user = new User();
    }
    
    public function index() {
        if ($this->user->isLoggedIn()) {
            header('Location: index.php?page=dashboard');
            exit;
        }
        include __DIR__ . '/../views/login.php';
    }
    
    public function proses() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $username = trim($_POST['username'] ?? '');
            $password = trim($_POST['password'] ?? '');
            
            if (empty($username) || empty($password)) {
                $_SESSION['error'] = 'Username dan Password harus diisi!';
                header('Location: index.php?page=login');
                exit;
            }
            
            if ($this->user->login($username, $password)) {
                $_SESSION['success'] = 'Selamat datang, ' . htmlspecialchars($username) . '!';
                header('Location: index.php?page=dashboard');
            } else {
                $_SESSION['error'] = 'Username atau Password salah!';
                header('Location: index.php?page=login');
            }
            exit;
        }
    }
    
    public function logout() {
        $this->user->logout();
        header('Location: index.php?page=login');
        exit;
    }
}
?>

<?php
// ============================================
// MENUITEM.PHP - MODEL BARANG (OOP)
// ============================================

require_once __DIR__ . '/../config/Database.php';

class MenuItem {
    private $db;
    
    public function __construct() {
        $this->db = Database::getInstance();
    }
    
    public function getAll($categoryFilter = null, $availableOnly = false) {
        $sql = "SELECT * FROM menu_items";
        $conditions = [];
        $params = [];
        
        if ($availableOnly) {
            $conditions[] = "is_available = 1";
        }
        if ($categoryFilter) {
            $conditions[] = "category = ?";
            $params[] = $categoryFilter;
        }
        if ($conditions) {
            $sql .= " WHERE " . implode(" AND ", $conditions);
        }
        $sql .= " ORDER BY category, item_name";
        return $this->db->getAll($sql, $params);
    }
    
    public function getById($id) {
        $sql = "SELECT * FROM menu_items WHERE item_id = ?";
        return $this->db->getOne($sql, [$id]);
    }
    
    public function getCategories() {
        $sql = "SELECT DISTINCT category FROM menu_items WHERE category IS NOT NULL ORDER BY category";
        $results = $this->db->getAll($sql);
        return array_column($results, 'category');
    }
    
    public function save($data) {
        if (isset($data['item_id']) && $data['item_id'] > 0) {
            $sql = "UPDATE menu_items SET item_code=?, item_name=?, category=?, price=?, is_available=? WHERE item_id=?";
            return $this->db->execute($sql, [
                $data['item_code'], $data['item_name'], $data['category'],
                $data['price'], $data['is_available'] ?? 1, $data['item_id']
            ]);
        } else {
            $sql = "INSERT INTO menu_items (item_code, item_name, category, price, is_available) VALUES (?, ?, ?, ?, ?)";
            return $this->db->insert($sql, [
                $data['item_code'], $data['item_name'], $data['category'],
                $data['price'], $data['is_available'] ?? 1
            ]);
        }
    }
    
    public function delete($id) {
        $sql = "DELETE FROM menu_items WHERE item_id = ?";
        return $this->db->execute($sql, [$id]);
    }
}

<?php
// ============================================
// ORDER.PHP - MODEL ORDER (OOP)
// ============================================

require_once __DIR__ . '/../config/Database.php';

class Order {
    private $db;
    
    public function __construct() {
        $this->db = Database::getInstance();
    }
    
    // STORED PROCEDURE: sp_create_order
    public function createOrder($table_id, $waiter_id) {
        try {
            $result = $this->db->callProcedure('sp_create_order', [$table_id, $waiter_id]);
            if ($result && isset($result[0][0]['order_id'])) {
                return $result[0][0]['order_id'];
            }
            return null;
        } catch (Exception $e) {
            throw new Exception("Gagal membuat order: " . $e->getMessage());
        }

    public function login($username, $password) {
        $sql = "SELECT * FROM users WHERE username = ? AND password = MD5(?) AND is_active = 1";
        $result = $this->db->getOne($sql, [$username, $password]);
        
        if ($result) {
            $this->user_id = $result['user_id'];

            $this->full_name = $result['full_name'];
            $this->role = $result['role'];
            
            $_SESSION['user_id'] = $result['user_id'];
            $_SESSION['username'] = $result['username'];
            $_SESSION['full_name'] = $result['full_name'];
            $_SESSION['role'] = $result['role'];
            $_SESSION['login'] = true;

            $params = session_get_cookie_params();
            setcookie(session_name(), '', time() - 42000,

                $params["secure"], $params["httponly"]
            );

        return true;

<script>
let items = [];
let itemCount = 0;

$(document).ready(function() {
    // Update subtotal ketika quantity berubah
    $('#quantity').on('change keyup', function() {
        updateSubtotal();
    });
    
    $('#id_barang').on('change', function() {
        updateSubtotal();
    });
    
    // Tambah item
    $('#btnAddItem').on('click', function() {
        addItem();
    });
    
    // Simpan order
    $('#btnSaveOrder').on('click', function() {
        saveOrder();
    });
});

function updateSubtotal() {
    let harga = $('#id_barang option:selected').data('harga') || 0;
    let qty = parseInt($('#quantity').val()) || 0;
    let subtotal = harga * qty;
    $('#subtotal').val('Rp ' + subtotal.toLocaleString('id-ID'));
}

function addItem() {
    let id_barang = $('#id_barang').val();
    let nama_barang = $('#id_barang option:selected').text().split(' (')[0];
    let harga = $('#id_barang option:selected').data('harga') || 0;
    let stok = $('#id_barang option:selected').data('stok') || 0;
    let qty = parseInt($('#quantity').val()) || 0;
    
    if (!id_barang) {
        alert('Pilih menu terlebih dahulu!');
        return;
    }
    
    if (qty <= 0) {
        alert('Quantity harus lebih dari 0!');
        return;
    }
    
    if (qty > stok) {
        alert('Stok tidak cukup! Sisa stok: ' + stok);
        return;
    }
    
    let subtotal = harga * qty;
    
    items.push({
        id_barang: id_barang,
        nama_barang: nama_barang,
        quantity: qty,
        harga: harga,
        subtotal: subtotal
    });
    
    renderItems();
    updateSummary();
    
    // Reset form
    $('#id_barang').val('');
    $('#quantity').val(1);
    $('#subtotal').val('0');
}

function renderItems() {
    let html = '';
    items.forEach(function(item, index) {
        html += `
            <tr>
                <td>${index + 1}</td>
                <td>${item.nama_barang}</td>
                <td>${item.quantity}</td>
                <td>Rp ${item.harga.toLocaleString('id-ID')}</td>
                <td>Rp ${item.subtotal.toLocaleString('id-ID')}</td>
                <td>
                    <button class="btn btn-sm btn-danger" onclick="removeItem(${index})">
                        <i class="fas fa-times"></i>
                    </button>
                </td>
            </tr>
        `;
    });
    $('#itemBody').html(html);
}

function removeItem(index) {
    items.splice(index, 1);
    renderItems();
    updateSummary();
}

function updateSummary() {
    let total = 0;
    items.forEach(function(item) {
        total += item.subtotal;
    });
    
    let meja = $('#id_meja option:selected').text() || '-';
    $('#summaryMeja').text(meja);
    $('#summaryItem').text(items.length);
    $('#summaryTotal').text('Rp ' + total.toLocaleString('id-ID'));
    $('#totalDisplay').text('Rp ' + total.toLocaleString('id-ID'));
}

function saveOrder() {
    let id_meja = $('#id_meja').val();
    
    if (!id_meja) {
        alert('Pilih meja terlebih dahulu!');
        return;
    }
    
    if (items.length === 0) {
        alert('Tambah item terlebih dahulu!');
        return;
    }
    
    if (!confirm('Yakin akan menyimpan order ini?')) {
        return;
    }
    
    $.ajax({
        url: 'index.php?page=order&action=save',
        type: 'POST',
        data: {
            id_meja: id_meja,
            items: JSON.stringify(items)
        },
        success: function(response) {
            alert('Order berhasil disimpan!');
            location.reload();
        },
        error: function() {
            alert('Gagal menyimpan order!');
        }
    });
}

-- ============================================================
-- DATABASE: restaurant_pos
-- SCRIPT LENGKAP: TABEL + DATA + STORED PROCEDURE

-- 1. BUAT DATABASE
-- ============================================================
CREATE DATABASE IF NOT EXISTS restaurant_pos;
USE restaurant_pos;

    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    role ENUM('administrator', 'waiter', 'kasir') NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- TABEL 2: tables (Meja)
CREATE TABLE tables (
    table_id INT AUTO_INCREMENT PRIMARY KEY,
    table_number VARCHAR(10) UNIQUE NOT NULL,
    capacity INT DEFAULT 4,
    status ENUM('available', 'occupied', 'reserved') DEFAULT 'available',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- TABEL 3: menu_items (Barang/Menu)
CREATE TABLE menu_items (
    item_id INT AUTO_INCREMENT PRIMARY KEY,
    item_code VARCHAR(20) UNIQUE NOT NULL,
    item_name VARCHAR(100) NOT NULL,
    category VARCHAR(50),
    price DECIMAL(10, 2) NOT NULL,
    is_available BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- TABEL 4: orders (Pesanan)
CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    table_id INT NOT NULL,
    waiter_id INT NOT NULL,
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status ENUM('pending', 'preparing', 'ready', 'served', 'paid') DEFAULT 'pending',
    subtotal DECIMAL(10, 2) DEFAULT 0,
    tax DECIMAL(10, 2) DEFAULT 0,
    total DECIMAL(10, 2) DEFAULT 0,
    FOREIGN KEY (table_id) REFERENCES tables(table_id),
    FOREIGN KEY (waiter_id) REFERENCES users(user_id)
);

-- TABEL 5: order_items (Detail Pesanan)
CREATE TABLE order_items (
    order_item_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    item_id INT NOT NULL,
    quantity INT NOT NULL,
    price_per_unit DECIMAL(10, 2) NOT NULL,
    subtotal DECIMAL(10, 2) NOT NULL,
    note TEXT,
    FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE,
    FOREIGN KEY (item_id) REFERENCES menu_items(item_id)
);

-- TABEL 6: transactions (Transaksi)
CREATE TABLE transactions (
    transaction_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    kasir_id INT NOT NULL,
    transaction_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    payment_method ENUM('cash', 'credit_card', 'debit_card', 'qr_code') NOT NULL,
    amount_paid DECIMAL(10, 2) NOT NULL,

    status ENUM('completed', 'cancelled') DEFAULT 'completed',
    FOREIGN KEY (order_id) REFERENCES orders(order_id),
    FOREIGN KEY (kasir_id) REFERENCES users(user_id)

-- SP 1: sp_create_order - Membuat order baru
CREATE PROCEDURE sp_create_order(
    IN p_table_id INT,
    IN p_waiter_id INT,
    OUT p_order_id INT
)
BEGIN
    INSERT INTO orders (table_id, waiter_id, status) 
    VALUES (p_table_id, p_waiter_id, 'pending');
    
    SET p_order_id = LAST_INSERT_ID();
    UPDATE tables SET status = 'occupied' WHERE table_id = p_table_id;
END //

-- SP 2: sp_add_order_item - Menambah item ke order
CREATE PROCEDURE sp_add_order_item(
    IN p_order_id INT,
    IN p_item_id INT,
    IN p_quantity INT,
    IN p_note TEXT
)
BEGIN
    DECLARE v_price DECIMAL(10,2);
    DECLARE v_subtotal DECIMAL(10,2);
    DECLARE v_current_subtotal DECIMAL(10,2);
    DECLARE v_tax DECIMAL(10,2);
    DECLARE v_total DECIMAL(10,2);
    
    -- Ambil harga item
    SELECT price INTO v_price FROM menu_items WHERE item_id = p_item_id;
    
    -- Hitung subtotal
    SET v_subtotal = v_price * p_quantity;
    
    -- Insert ke order_items
    INSERT INTO order_items (order_id, item_id, quantity, price_per_unit, subtotal, note)
    VALUES (p_order_id, p_item_id, p_quantity, v_price, v_subtotal, p_note);
    
    -- Update total di orders
    SELECT SUM(subtotal) INTO v_current_subtotal FROM order_items WHERE order_id = p_order_id;
    
    SET v_tax = v_current_subtotal * 0.1; -- PPN 10%
    SET v_total = v_current_subtotal + v_tax;
    
    UPDATE orders 
    SET subtotal = v_current_subtotal, tax = v_tax, total = v_total 
    WHERE order_id = p_order_id;
END //

-- SP 3: sp_complete_transaction - Menyelesaikan transaksi
CREATE PROCEDURE sp_complete_transaction(
    IN p_order_id INT,
    IN p_kasir_id INT,
    IN p_payment_method VARCHAR(20),
    IN p_amount_paid DECIMAL(10,2)
)
BEGIN
    DECLARE v_total DECIMAL(10,2);
    DECLARE v_change DECIMAL(10,2);
    DECLARE v_table_id INT;
    
    -- Ambil total order
    SELECT total, table_id INTO v_total, v_table_id 
    FROM orders WHERE order_id = p_order_id;
    
    -- Hitung kembalian
    SET v_change = p_amount_paid - v_total;
    
    -- Insert transaksi
    INSERT INTO transactions (order_id, kasir_id, payment_method, amount_paid, change_amount)
    VALUES (p_order_id, p_kasir_id, p_payment_method, p_amount_paid, v_change);
    
    -- Update status order menjadi paid
    UPDATE orders SET status = 'paid' WHERE order_id = p_order_id;
    
    -- Update status meja menjadi available
    UPDATE tables SET status = 'available' WHERE table_id = v_table_id;
END //

-- SP 4: sp_daily_report - Laporan penjualan harian
CREATE PROCEDURE sp_daily_report(
    IN p_date DATE
)
BEGIN
    SELECT 
        t.transaction_id,
        t.transaction_date,
        u.full_name AS kasir_name,
        o.order_id,
        t.payment_method,
        o.total,
        t.amount_paid,
        t.change_amount
    FROM transactions t
    JOIN orders o ON t.order_id = o.order_id
    JOIN users u ON t.kasir_id = u.user_id
    WHERE DATE(t.transaction_date) = p_date
    ORDER BY t.transaction_date DESC;
END //

-- SP 5: sp_best_selling_menu - Laporan menu terlaris
CREATE PROCEDURE sp_best_selling_menu(
    IN p_start_date DATE,
    IN p_end_date DATE
)
BEGIN
    SELECT 
        mi.item_name,
        mi.category,
        SUM(oi.quantity) AS total_quantity,
        SUM(oi.subtotal) AS total_revenue
    FROM order_items oi
    JOIN orders o ON oi.order_id = o.order_id
    JOIN menu_items mi ON oi.item_id = mi.item_id
    WHERE DATE(o.order_date) BETWEEN p_start_date AND p_end_date
    AND o.status = 'paid'
    GROUP BY mi.item_id
    ORDER BY total_quantity DESC
    LIMIT 10;
END //

DELIMITER ;
