```php
<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, ...$roles)
    {
        if (!session()->has('user_id') || !session()->has('role')) {
            return redirect()->route('login');
        }

        if (!in_array(session('role'), $roles)) {
            abort(403, 'Akses ditolak. Role Anda tidak diizinkan.');
        }

        return $next($request);
    }
}
```

```php
<?php

namespace App\Http\Controllers;

use App\Models\Masyarakat;
use App\Models\Petugas;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;

class AuthController extends Controller
{
    // ============ LOGIN ============
    public function showLogin()
    {
        if (session()->has('user_id')) {
            return redirect()->route('dashboard');
        }
        return view('auth.login');
    }

    public function login(Request $request)
    {
        $request->validate([
            'username' => 'required',
            'password' => 'required',
            'role' => 'required|in:administrator,petugas,masyarakat',
        ]);

        $user = null;
        $roleName = '';

        if (in_array($request->role, ['administrator', 'petugas'])) {
            $user = Petugas::where('username', $request->username)->first();
            if ($user) {
                $roleName = ($user->id_level == 1) ? 'administrator' : 'petugas';
            }
        } else {
            $user = Masyarakat::where('username', $request->username)->first();
            if ($user) $roleName = 'masyarakat';
        }

        if (!$user || !Hash::check($request->password, $user->password)) {
            return back()->withErrors(['username' => 'Username atau password salah!'])->withInput();
        }

        if ($request->role !== $roleName) {
            return back()->withErrors(['username' => 'Role tidak sesuai dengan akun Anda!'])->withInput();
        }

        session([
            'user_id' => $user->id_petugas ?? $user->id_user,
            'nama' => $user->nama_petugas ?? $user->nama_lengkap,
            'username' => $user->username,
            'role' => $roleName,
        ]);

        return redirect()->route('dashboard')->with('success', 'Selamat datang, ' . session('nama') . '!');
    }

    // ============ REGISTER ============
    public function showRegister()
    {
        if (session()->has('user_id')) {
            return redirect()->route('dashboard');
        }
        return view('auth.register');
    }

    public function register(Request $request)
    {
        $request->validate([
            'role' => 'required|in:administrator,petugas,masyarakat',
            'nama' => 'required|min:3',
            'username' => 'required|min:3',
            'password' => 'required|min:6|confirmed',
            'telp' => 'nullable',
        ]);

        try {
            $hash = Hash::make($request->password);

            if ($request->role === 'masyarakat') {
                DB::statement("CALL sp_register_masyarakat(?, ?, ?, ?)", [
                    $request->nama,
                    $request->username,
                    $hash,
                    $request->telp ?? '',
                ]);
            } else {
                $idLevel = ($request->role === 'administrator') ? 1 : 2;
                DB::statement("CALL sp_register_petugas(?, ?, ?, ?)", [
                    $request->nama,
                    $request->username,
                    $hash,
                    $idLevel,
                ]);
            }

            return redirect()->route('login')->with('success', 'Registrasi berhasil! Silakan login.');
        } catch (\Exception $e) {
            return back()->withErrors(['username' => $e->getMessage()])->withInput();
        }
    }

    // ============ LOGOUT ============
    public function logout()
    {
        session()->flush();
        return redirect()->route('login');
    }
```

```php
        return view('lelang.form', compact('barang'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'id_barang' => 'required|exists:tb_barang,id_barang',
            'tgl_lelang' => 'required|date',
        ]);

        Lelang::create([
            'id_barang' => $request->id_barang,
            'tgl_lelang' => $request->tgl_lelang,
            'id_petugas' => session('user_id'),
            'status' => 'ditutup',
        ]);

        return redirect()->route('lelang.index')->with('success', 'Lelang berhasil dibuat! Klik "Buka" untuk memulai.');
    }

    public function show(Lelang $lelang)
    {
        $lelang->load(['barang', 'pemenang', 'petugas', 'history.user']);
        return view('lelang.show', compact('lelang'));
    }

    public function destroy(Lelang $lelang)
    {
        $lelang->delete();
        return redirect()->route('lelang.index')->with('success', 'Lelang berhasil dihapus!');
    }

    public function buka($id)
    {
        try {
            DB::statement("CALL sp_buka_lelang(?)", [$id]);
            return redirect()->route('lelang.index')->with('success', 'Lelang dibuka!');
        } catch (\Exception $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    public function tutup($id)
    {
        try {
            DB::statement("CALL sp_tutup_lelang(?)", [$id]);
            return redirect()->route('lelang.index')->with('success', 'Lelang ditutup! Pemenang sudah ditentukan.');
        } catch (\Exception $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    // ============ UNTUK MASYARAKAT ============
    public function aktif()
    {
        $lelang = Lelang::with('barang')
            ->where('status', 'dibuka')
            ->orderByDesc('id_lelang')
            ->get();

        foreach ($lelang as $l) {
            $l->penawaran_tertinggi = HistoryLelang::where('id_lelang', $l->id_lelang)->max('penawaran_harga');
        }

        return view('lelang.aktif', compact('lelang'));
    }

    public function tawar(Request $request)
    {
        $request->validate([
            'id_lelang' => 'required|exists:tb_lelang,id_lelang',
            'id_barang' => 'required|exists:tb_barang,id_barang',
            'penawaran_harga' => 'required|integer|min:1',
        ]);

        try {
            DB::statement("CALL sp_tambah_penawaran(?, ?, ?, ?)", [
                $request->id_lelang,
                $request->id_barang,
                session('user_id'),
                $request->penawaran_harga,
            ]);

            return back()->with('success', 'Penawaran berhasil! Rp ' . number_format($request->penawaran_harga, 0, ',', '.'));
        } catch (\Exception $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    public function riwayatSaya()
    {
        $history = HistoryLelang::with(['barang', 'lelang'])
            ->where('id_user', session('user_id'))
            ->latest()
            ->get();

        return view('lelang.riwayat', compact('history'));
    }
```
